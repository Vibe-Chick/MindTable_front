# --- 환경 변수 (.env) ---
# VITE_API_BASE_URL=/api          백엔드 API 주소 (Vite 프록시를 거치는 상대 경로)
# VITE_PROXY_TARGET=              /api 를 전달할 백엔드 원본. 비우면 http://127.0.0.1:8000, 배포 서버면 https://xxx.run.app
# VITE_USE_MOCK=true              true면 백엔드 없이 mock 데이터로 동작
# VITE_LIVE_APIS=auth/google      mock 모드에서도 실제 백엔드로 보낼 API 목록 (쉼표 구분)
# VITE_GOOGLE_CLIENT_ID=          Google Cloud 콘솔 OAuth 클라이언트 ID (mock=false일 때 필요)
# VITE_IDLE_MINUTES=30           이 시간 동안 활동 없으면 자동 로그아웃 (기본 30, 테스트 시 1)
# VITE_VAPID_PUBLIC_KEY=          Web Push VAPID 공개키 (백엔드가 생성해서 공유, mock 모드에선 불필요)

# --- 백엔드 (MindTable_back, Django + DRF + SimpleJWT, http://127.0.0.1:8000 — Vite 프록시로 /api 전달) ---
# 화면별 정리는 Notion "API 명세서"와 동일. 토큰 O = Authorization: Bearer <access> 필요 (src/service/api.js 인터셉터가 자동 첨부)
#
# [회원가입]
# POST  /api/auth/google/              토큰 X   { credential }  → { access, refresh, user{id,email,name,picture,google_id}, created }   [Done]
# POST  /api/auth/school/send-code/    토큰 O   { school_email }  → { message }                                                       [Done]
# POST  /api/auth/token/refresh/       토큰 X   { refresh }  → { access }   access 만료(401) 시 프론트가 자동 호출 후 원래 요청 재시도 (SimpleJWT TokenRefreshView)   [백엔드 미구현]
# POST  /api/auth/school/verify-code/  토큰 O   { school_email, code, school_name, major }  → { school_verification: { school_email, school_name, verified_at } }   [Done]
#
# [심리 테스트]  (질문은 백엔드가 문항마다 LLM으로 생성, id 는 백엔드 내부 관리 → 프론트는 질문 문자열·답변만 주고받음)
# POST  /api/psychology/questions/     토큰 O   {}  → { questions: '질문 문자열' }   문항마다 호출(3회), 호출할 때마다 다른 질문
# POST  /api/psychology/check-answer/  토큰 O   { question, answer }  → { needFollowUp, followUpQuestion|null }   답변 검증 + 애매하면 꼬리 질문 생성 (테스트당 최대 1개 = 4번째 문항)
# POST  /api/psychology/analyze/       토큰 O   { user_id: <이메일>, followUps: [{ questionId, question, answer }] }  → { bigFive, interests, summary, valid, insufficient[] }   AI 성향 분석 (기본 답변은 check-answer 때 백엔드가 저장)
# POST  /api/users/{id}/profile/       토큰 O   profile  → { ok }                                                분석된 프로필 저장
#
# [매칭]
# POST  /api/match/request/            토큰 O   { userId, slots: ['2026-09-21-lunch', ...] | ['now'] }  → { matchId }   (희망 시간대)
# GET   /api/match/{id}/               토큰 O   → { status, result: { ..., mealAt } }   (mealAt: 확정된 식사 시각, 리뷰 창 계산용)
#
# [식당 추천]
# GET   /api/match/{id}/restaurants/               토큰 O   ?location&price&avoid  → { source: 'partner'|'map', restaurants[] }   (제휴 없으면 지도 API fallback)
# POST  /api/match/{id}/reserve/                   토큰 O   { restaurantId }
# POST  /api/match/{id}/restaurants/preferences/   토큰 O   { location, customLocation, price, avoid }  → { submitted, total, common }
# GET   /api/match/{id}/vote/                      토큰 O   → { votes: {userId: restaurantId}, confirmed: restaurantId|null }
# POST  /api/match/{id}/vote/                      토큰 O   { restaurantId }  → 위와 동일 (과반이면 confirmed)
#
# [마이페이지]
# PATCH /api/users/{id}/               토큰 O   { ...patch }  → user            내 정보 수정
# GET   /api/users/{id}/matches/       토큰 O   → history[]                     매칭 기록
#
# [테이블 리뷰]
# GET   /api/match/{id}/review/        토큰 O   → { submitted, result }                                                    리뷰 제출 상태
# POST  /api/match/{id}/review/        토큰 O   { answers }  → { deltas, before, after, diversityPref, summary, driftAlert }   테이블 리뷰 제출 → 프로필 반영
# GET   /api/users/{id}/reviews/       토큰 O   → [{ matchId, date, restaurant, deltas, diversityPref, driftAlert }]        내 리뷰 기록
#
# [푸시 알림]  (백엔드: pywebpush + VAPID 키 필요)
# POST   /api/push/subscribe/          토큰 O   PushSubscription.toJSON() = { endpoint, keys: { p256dh, auth } }  → 사용자·기기별 저장
# DELETE /api/push/subscribe/          토큰 O   { endpoint }
# PATCH  /api/push/preferences/        토큰 O   { enabled, match_done, place_confirmed, review_request }
#   발송 시점: 매칭 완료 · 투표로 장소 확정 · 식사 2시간 후 리뷰 요청
#   payload: { title, body, url, tag? }  → public/sw.js 가 OS 알림으로 표시, 탭하면 url 로 이동
#
# ※ {id} 경로는 백엔드에서 토큰의 user와 {id} 일치 여부를 검사할 것
# ※ 모든 경로는 끝 슬래시(/)로 통일 (Django APPEND_SLASH — POST/PATCH는 슬래시 없으면 실패)
