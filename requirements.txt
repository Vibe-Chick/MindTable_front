# --- 백엔드 (MindTable_back, Django, http://127.0.0.1:8000 — Vite 프록시로 /api 전달) ---
# POST  /auth/google/             { credential }  → { access, refresh, user{id,email,name,picture,google_id}, created }   [연동됨]
# POST  /auth/school/code         { userId, univEmail }  → { school }
# POST  /auth/school/verify       { userId, univEmail, code, school, major }  → user
# PATCH /users/:id                { ...patch }  → user
# POST  /profile/check-answer     { questionId, trait, question, answer }  → { ok, reason }
# POST  /profile/analyze          { answers }  → { bigFive, interests, summary, valid, insufficient[] }
# POST  /users/:id/profile        profile
# POST  /match/request            { userId }  → { matchId }
# GET   /match/:id                → { status, result }
# GET   /users/:id/matches        → history[]
# POST  /match/:id/restaurants/preferences/  { location, customLocation, price, avoid }  → { submitted, total, common }
# GET   /match/:id/restaurants/   ?location&price&avoid  → { source: 'partner'|'map', restaurants[] }   (제휴 없으면 지도 API fallback)
# GET   /match/:id/vote/          → { votes: {userId: restaurantId}, confirmed: restaurantId|null }
# POST  /match/:id/vote/          { restaurantId }  → 위와 동일 (과반이면 confirmed)
# POST  /match/:id/reserve        { restaurantId }
# GET   /match/:id/review/        → { submitted, result }                          (식사 후 리뷰 상태)
# POST  /match/:id/review/        { answers }  → { deltas, before, after, diversityPref, summary, driftAlert }   (보정)
# --- Web Push (백엔드: pywebpush + VAPID 키 필요) ---
# POST   /push/subscribe/         PushSubscription.toJSON() = { endpoint, keys: { p256dh, auth } }  → 사용자·기기별 저장
# DELETE /push/subscribe/         { endpoint }
# PATCH  /push/preferences/       { enabled, match_done, place_confirmed, review_request }
#   발송 시점: 매칭 완료 · 투표로 장소 확정 · 식사 2시간 후 리뷰 요청
#   payload: { title, body, url, tag? }  → public/sw.js 가 OS 알림으로 표시, 탭하면 url 로 이동
